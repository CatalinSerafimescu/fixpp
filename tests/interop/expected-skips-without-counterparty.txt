# result=SKIPPED, with a skip message matching the regex
#
#   quickfix-(cpp|j) unavailable: INTEROP_QUICKFIX_(CPP|J)_PORT not set
#   \(parent harness did not lease a port\)
#
# whenever `ctest -L interop` runs with neither INTEROP_QUICKFIX_CPP_PORT nor
# INTEROP_QUICKFIX_J_PORT set in the environment — the #431 gate step in each
# tier workflow `unset`s both (and their _HOST siblings) before running, so
# this file's condition is established by the step rather than assumed of the
# lane it runs on. `ci/assert-interop-skips.py` enforces both directions: an
# id that skips here and is NOT listed, and a listed id that instead runs (or
# fails, or skips for a different reason this time), are both violations.
#
# A case reaching this list got there via `INTEROP_REQUIRE_COUNTERPARTY(...)`
# (tests/interop/support/counterparty_probe.hpp) and nothing past that guard
# — a test that also has a SECOND, later skip guard (e.g. a
# `FIXPP_TLS_FIXTURE_DIR`-not-set or `skip:not-applicable` disposition) never
# reaches this reason and must not be listed here for that combo.
#
# FORMAT: one fully-qualified gtest id per line, `<TestSuiteName>.<TestName>`

# lines ignored.
#

# set in the environment:
#
#   GTEST_OUTPUT=json:<dir>/ ctest --preset <preset> -L interop \
#       --output-on-failure --no-tests=error --timeout 1800
#   python3 - <<'PY'
#   import glob, json
#   ids = set()
#   for f in glob.glob("<dir>/*.json"):
#       doc = json.load(open(f))
#       for ts in doc["testsuites"]:
#           for tc in ts["testsuite"]:
#               if tc.get("result") == "SKIPPED":
#                   ids.add(f"{ts['name']}.{tc['name']}")
